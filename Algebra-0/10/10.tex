\begin{problem}[7E-8]
设 $T \in \mathcal{L}(V, W)$. 设 $s_1 \ge s_2 \ge \cdots \ge s_m > 0$, $e_1, \ldots, e_m$ 是 $V$ 中的规范正交组, $f_1, \ldots, f_m$ 是 $W$ 中的规范正交组, 使得
\[
T v = s_1 \langle v, e_1 \rangle f_1 + \cdots + s_m \langle v, e_m \rangle f_m
\]
对任一 $v \in V$ 都成立.
\begin{enumerate}[label=(\alph*)]
\item 证明: $f_1, \ldots, f_m$ 是 $\operatorname{range} T$ 的规范正交基.
\item 证明: $e_1, \ldots, e_m$ 是 $(\operatorname{null} T)^\perp$ 的规范正交基.
\item 证明: $s_1, \ldots, s_m$ 是 $T$ 的正奇异值.
\item 证明: 如果 $k \in \{1, \ldots, m\}$, 那么 $e_k$ 是 $T^* T$ 的特征向量, 对应特征值是 $s_k^2$.
\item 证明:
\[
T T^* w = s_1^2 \langle w, f_1 \rangle f_1 + \cdots + s_m^2 \langle w, f_m \rangle f_m
\]
对所有 $w \in W$ 成立.
\end{enumerate}
\end{problem}

\begin{problem}[7E-11]
设 $T \in \mathcal{L}(V, W)$, $v_1, \ldots, v_n$ 是 $V$ 的规范正交基. 令 $s_1, \ldots, s_n$ 表示 $T$ 的奇异值.
\begin{enumerate}[label=(\alph*)]
\item 证明: $\|T v_1\|^2 + \cdots + \|T v_n\|^2 = s_1^2 + \cdots + s_n^2$.
\item 证明: 如果 $W = V$ 且 $T$ 是正算子, 那么 $\langle T v_1, v_1 \rangle + \cdots + \langle T v_n, v_n \rangle = s_1 + \cdots + s_n$.
\end{enumerate}
\end{problem}

\begin{problem}[7F-27]
设 $T \in \mathcal{L}(V)$, $s_1, \ldots, s_n$ 是 $T$ 的奇异值. 令 $e_1, \ldots, e_n$ 和 $f_1, \ldots, f_n$ 都是 $V$ 的规范正交基, 使得
\[
T v = s_1 \langle v, e_1 \rangle f_1 + \cdots + s_n \langle v, e_n \rangle f_n
\]
对所有 $v \in V$ 成立. 定义 $S \in \mathcal{L}(V)$ 为
\[
S v = \langle v, e_1 \rangle f_1 + \cdots + \langle v, e_n \rangle f_n.
\]
\begin{enumerate}[label=(\alph*)]
\item 证明: $S$ 是么正的且 $\|T - S\| = \max\{ |s_1 - 1|, \ldots, |s_n - 1| \}$.
\item 证明: 如果 $E \in \mathcal{L}(V)$ 是么正的, 那么 $\|T - E\| \ge \|T - S\|$.
\end{enumerate}

注: 本题所求的是, (在幺正算子当中) 尽可能接近给定算子 $T$ 的幺正算子 $S$.
\end{problem}

\begin{problem}[7F-30]
设 $T \in \mathcal{L}(V)$, 而 $S \in \mathcal{L}(V)$ 是幺正算子, $R \in \mathcal{L}(V)$ 是使得 $T = S R$ 的正算子. 证明: $R = \sqrt{T^* T}$.

注: 本题表明, 如果我们将 $T$ 写成一个幺正算子和一个正算子的乘积 (如 7.93 的极分解所示), 那么其中的正算子就等于 $\sqrt{T^* T}$.
\end{problem}
