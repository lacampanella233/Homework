\begin{problem}
  {\heiti 恒等态射与逆态射}(\textbf{identity and inverse morphisms}). 设 $\mathcal C$ 是一个范畴.

  \begin{enumerate}
    \item[(a)] 设
      \[
        f:X\to Y,\qquad r,s:Y\to X
      \]
      满足
      \[
        r\circ f=\id_X,\qquad f\circ s=\id_Y.
      \]
      此时分别称 $r$ 为 $f$ 的左逆(left inverse), $s$ 为 $f$ 的右逆(right inverse). 证明 $r=s$, 并由此证明逆态射的唯一性. 进一步, 若 $X\xrightarrow{f}Y\xrightarrow{g}Z$ 均为同构(isomorphisms), 证明
      \[
        (g\circ f)^{-1}=f^{-1}\circ g^{-1}.
      \]

    \item[(b)] 在集合范畴(category of sets) $\mathbf{Set}$ 中, 令
      \[
        i:\{0\}\longrightarrow\{0,1\},\qquad i(0)=0,
      \]
      并令 $r:\{0,1\}\to\{0\}$ 为唯一的映射. 计算 $r\circ i$ 和 $i\circ r$, 说明为什么``有左逆''和``有右逆''都不能单独作为同构的定义.
  \end{enumerate}
\end{problem}

\begin{solution}
	(a) 根据恒等态射的定义以及结合律,
	\[ r = r \circ \id_Y = r \circ (f \circ s)
	= (r \circ f) \circ s = \id_X \circ s = s. \]
	因此, 如果 $f\in \Hom_{\mathcal{C}}(X,Y)$ 有两个逆, 记为 $g_1, g_2 \in \Hom_{\mathcal{C}}(Y,X)$, 则有 $f \circ g_1 = \id_Y$, 并且 $g_2 \circ f = \id_X$, 因此 $g_1=g_2$.

	下面来证明 $(g\circ f)^{-1} = f^{-1}\circ g^{-1}$. 直接的计算给出
	\begin{align*}
		(g\circ f) \circ (f^{-1}\circ g^{-1})
		&= g \circ (f\circ f^{-1}) \circ g^{-1} 
		= g\circ \id_Y \circ g^{-1}
		= g\circ g^{-1} = \id_Z,\\
		(f^{-1}\circ g^{-1}) \circ (g\circ f) 
		&= f^{-1} \circ (g^{-1} \circ g) \circ f
		= f^{-1} \circ \id_Y \circ f
		= f^{-1} \circ f = \id_X.
	\end{align*}

	(b) 计算表明: $(r \circ i) (0) = 0$, 而
	\[ (i \circ r) (0) = (i \circ r) (1) = 0. \]
	此时, $i$ 的左逆为 $r$, $r$ 的右逆为 $i$, 但是 $i$ 和 $r$ 均不是对方的逆态射.
\end{solution}



\begin{problem}
  {\heiti 由关系构造范畴}(\textbf{categories associated with relations}). 设 $S$ 是集合, $R\subseteq S\times S$ 是一个关系(relation). 考虑以下候选范畴 $\mathcal C_R$: 对象为 $S$ 的元素, 并规定
  \[
    \Hom_{\mathcal C_R}(x,y)=
    \begin{cases}
      \{u_{x,y}\}, & xRy,\\
      \varnothing, & \text{其他情况}.
    \end{cases}
  \]
  复合预定为 $u_{y,z}\circ u_{x,y}=u_{x,z}$.

  \begin{enumerate}
    \item[(a)] 证明: 上述数据能够定义一个范畴, 当且仅当 $R$ 满足自反性(reflexivity)和传递性(transitivity). 写出恒等态射, 并验证结合律(associativity).

    \item[(b)] 假设上述数据定义了范畴. 证明
      \[
        x\cong y\qquad\Longleftrightarrow\qquad xRy\mathbin{\text{ 且 }}yRx,
      \]
      其中 $x\cong y$ 表示对象 $x$ 与 $y$ 之间存在同构. 进一步证明, $\mathcal C_R$ 是群胚, 当且仅当 $R$ 还满足对称性(symmetry). 因此, 等价关系(equivalence relation)给出群胚, 而偏序(partial order)给出的范畴中两个对象同构当且仅当它们相等.

    \item[(c)] 分别计算下列范畴的态射总数、对象的同构类, 以及每个对象的自同构群(automorphism group):
      \begin{enumerate}
        \item[(i)] $S=\{1,2,3,6\}$, 关系为整除(divisibility);
        \item[(ii)] $S=\{0,1,2,3\}$, 关系为``具有相同的奇偶性''.
      \end{enumerate}
  \end{enumerate}
\end{problem}

\begin{solution}
	(a) 先来证明: 如果 $R$ 满足自反性和传递性, 则 $\mathcal{C}_R$ 是范畴. 由于 $xRx$, 因此 $\Hom_{\mathcal{C}_R}(x,x) = \{u_{x,x}\}$, 即恒等态射存在; 由于 $xRy, yRz \implies xRz$, 因此若 $u_{x,y}$ 和 $u_{y,z}$ 存在, 则 $u_{x,z} = u_{y,z} \circ u_{x,y}$ 存在. 并且, 由复合的定义, 结合律和恒等态射的性质天然成立(详细见下文). 因此 $\mathcal{C}_R$ 是范畴.

	再来证明: 若 $\mathcal{C}_R$ 是一个范畴, 则 $R$ 满足自反性和传递性. 由于对于任意一个对象 $x$, 均有 $\id_x \in \Hom_{\mathcal{C}_R}(x,x)$, 因此后者非空, 因此 $xRx$, 自反性成立; 若 $xRy$ 且 $yRz$, 则 $u_{x,z} = u_{y,z} \circ u_{x,y}$ 存在, 即 $xRz$, 因此传递性成立.

	对于每一个对象 $x$, 其恒等态射即为 $u_{x,x}$. 对于满足 $xRy$, $yRz$, $zRw$ 的对象 $x,y,z,w$, 有
	\begin{align*}
		u_{z,w} \circ (u_{y,z} \circ u_{x,y})
		&= u_{z,w} \circ u_{x,z}
		= u_{x,w} \\
		&= u_{y,w} \circ u_{x,y}
		=(u_{z,w} \circ u_{y,z}) \circ u_{x,y}.
	\end{align*}
	因此结合律成立.

	(b) 先证明第一个命题. 若 $x,y$ 之间存在同构, 则 $\Hom_{\mathcal{C}_R}(x,y)$ 与 $\Hom_{\mathcal{C}_R}(y,x)$ 均非空, 即 $xRy$ 且 $yRx$; 反过来, 若 $xRy$ 且 $yRx$, 则 $u_{x,y}$ 与 $u_{y,x}$ 均存在. 而根据结合律, 	\[ u_{y,x} \circ u_{x,y} = u_{x,x} = \id_x,\quad 
	u_{x,y} \circ u_{y,x} = u_{y,y} = \id_y, \]
	因此 $u_{x,y} = u_{y,x}^{-1}$, 从而两者皆为同构, 因此 $x \cong y$.

	再证明第二个命题. 若 $\mathcal{C}_R$ 是群胚, 则每一个态射都是同构, 因此都有逆态射. 因此, 若 $xRy$, 则 $u_{x,y}$ 存在, 因此 $u_{y,x}$ 存在, 从而 $yRx$. 这说明 $R$ 具有对称性; 反过来, 若 $R$ 具有对称性, 则对于满足 $xRy$ 的对象 $x,y$, 由 $yRx$ 知 $u_{y,x}$ 存在, 即 $u_{x,y}$ 存在逆态射, 所以为同构. 因此 $\mathcal{C}_R$ 为群胚.

	(c) 这两个范畴的示意图如图 \ref{category_sketch} 所示. 第一个范畴拥有 $9$ 个态射, 每个对象单独为一个同构类; 第二个范畴拥有 $8$ 个态射, 同构类为 $\{0,2\}$ 和 $\{1,3\}$. 所有的对象的自同构群均为平凡群 $\{e\}$.

\begin{problem}
  {\heiti 相反范畴}(\textbf{opposite category}){\heiti 与群胚核心}(\textbf{groupoid core}).

  \begin{enumerate}
    \item[(a)] 对范畴 $\mathcal C$, 定义 $\mathcal C\op$ 的对象与 $\mathcal C$ 相同, 并令
      \[
        \Hom_{\mathcal C\op}(X,Y)=\Hom_{\mathcal C}(Y,X).
      \]
      将 $f:X\to Y$ 对应的态射记为 $f\op:Y\to X$. 对于 $\mathcal C$ 中的 $X\xrightarrow{f}Y\xrightarrow{g}Z$, 规定
      \[
        f\op\circ g\op=(g\circ f)\op,\qquad
        \id^{\mathcal C\op}_X=(\id^{\mathcal C}_X)\op.
      \]
      验证这些数据定义一个范畴, 并写出
      \[
        (\mathcal C\op)\op\cong\mathcal C
      \]
      在对象和态射上的对应. 解释为什么 $f\op$ 与 $f^{-1}$ 不是同一个概念.

    \item[(b)] 定义 $\mathcal C^{\simeq}$ 的对象与 $\mathcal C$ 相同, 并令
      \[
        \Hom_{\mathcal C^{\simeq}}(X,Y)=\operatorname{Isom}_{\mathcal C}(X,Y),
      \]
      其中右端是从 $X$ 到 $Y$ 的所有同构组成的集合; 复合与恒等态射继承自 $\mathcal C$. 验证 $\mathcal C^{\simeq}$ 是群胚. 进一步证明, 对任意对象 $X$, $\operatorname{Aut}_{\mathcal C}(X)$ 在复合运算下是一个群(group); 这里不需要假设 $\mathcal C$ 本身是群胚.
  \end{enumerate}
\end{problem}

\begin{solution}
	(a) 先验证 $\mathcal{C}\op$ 构成一个范畴. 

	对于每一个 $\mathcal{C}\op$ 中的对象 $X$, 按照上面的定义, 其恒等态射 $\id_X^{\mathcal{C}\op} = (\id_{X}^{\mathcal{C}})\op$ 存在; 两个态射的复合也由上面定义; 结合律由下面的计算验证: 对于 $\mathcal{C}$ 中的 $X \xrightarrow{f} Y \xrightarrow{g} Z \xrightarrow{h} W$,
	\begin{align*}
		(f\op \circ g\op) \circ h\op
		&= (g \circ f)\op \circ h\op
		= (h\circ (g\circ f))\op \\
		&= ((h\circ g) \circ f)\op 
		= f\op \circ (h\circ g)\op 
		= f\op \circ (g\op \circ h\op).
	\end{align*}
	恒等态射的性质由下面的计算验证: 对于 $f \in \Hom_{\mathcal{C}}(X,Y)$, 
	\begin{align*}
		f\op \circ \id_Y^{\mathcal{C}\op}
		&= (\id_Y^{\mathcal{C}} \circ f)\op
		= f\op, \\
		\id_X^{\mathcal{C}\op} \circ f\op
		&= (f \circ \id_X^{\mathcal{C}}) \op
		=f\op.
	\end{align*}

	现在来解释 $(\mathcal{C}\op)\op \cong \mathcal{C}$. 将 $\mathcal{C}$ 中的元素 $X$ 与 $(\mathcal{C}\op)\op$ 中的元素 $X$ 对应, 而将 $\mathcal{C}$ 中的态射 $f$ 与 $(\mathcal{C}\op)\op$ 中的态射 $(f\op)\op$ 对应. 则
	\[ (g\op)\op \circ (f\op)\op = (f\op \circ g\op)\op = ((g\circ f)\op)\op, \quad
	\id_X^{(\mathcal{C}\op)\op} = (\id_X^{\mathcal{C}\op})\op = ((\id_X^{\mathcal{C}})\op)\op. \]
	因此可以认为 $(\mathcal{C}\op)\op \cong \mathcal{C}$.

	由于 $\mathcal{C}$ 中的任意一个态射 $f$ 不一定存在逆态射 $f^{-1}$, 而在上述定义下一定存在 $f\op$, 因此两者不能视为一个概念.

	(b) 先验证 $C^{\simeq}$ 是一个群胚. 由于在定义中 $\mathcal{C}^{\simeq}$ 中的所有态射均为同构, 因此只需要验证 $\mathcal{C}^{\simeq}$ 构成一个范畴.

	对于任意的对象 $X$, 其恒等态射 $\id_X$ 是同构, 因此 $\id_X \in \operatorname{Isom}_{\mathcal{C}}(X,X) = \Hom_{\mathcal{C}^{\simeq}}(X,X)$, 因此 $\mathcal{C}^{\simeq}$ 中的任意对象存在恒等态射; 而对于 $\mathcal{C}^{\simeq}$ 中的 $X \xrightarrow{f} Y \xrightarrow{g} Z$, 由 $(g\circ f)^{-1} = f^{-1} \circ g^{-1}$, 知 $g\circ f$ 在 $\mathcal{C}$ 中是同构, 因此是 $\mathcal{C}^{\simeq}$ 中的态射. 这说明 $\mathcal{C}^{\simeq}$ 中的任意两个态射都存在复合. 因此 $\mathcal{C}^{\simeq}$ 构成一个范畴, 从而是一个群胚.

	下面来验证 $\Aut_\mathcal{C} (X)$ 构成一个群. 单位元 $\id_X$ 存在, 并且满足对于任意的 $f\in \Aut_{\mathcal{C}}(X)$, $f\circ \id_X = \id_X \circ f = f$; 对于任意的 $f,g\in \Aut_\mathcal{C}(X)$, 其复合 $g\circ f \in \Aut_{\mathcal{C}}(X)$, 并且满足结合律; 对于任意的 $f\in \Aut_{\mathcal{C}}(X)$, 由于其为同构, 因此存在逆态射 $f^{-1}\in \Aut_{\mathcal{C}}(X)$, 并且满足 $f \circ f^{-1} = f^{-1} \circ f = \id_X$. 因此 $\Aut_{\mathcal{C}}(X)$ 构成一个群. 
\end{solution}



\begin{problem}
  {\heiti 切片范畴、余切片范畴与对偶}(\textbf{duality}). 固定范畴 $\mathcal C$ 中的对象 $A$.

  定义切片范畴(slice category, 又称 overcategory) $\mathcal C/A$: 对象是态射 $p:X\to A$; 从 $p:X\to A$ 到 $q:Y\to A$ 的态射是满足 $q\circ u=p$ 的态射 $u:X\to Y$.

  定义余切片范畴(coslice category, 又称 undercategory) $A/\mathcal C$: 对象是态射 $p:A\to X$; 从 $p:A\to X$ 到 $q:A\to Y$ 的态射是满足 $u\circ p=q$ 的态射 $u:X\to Y$.

  \begin{enumerate}
    \item[(a)] 在上述两个构造中, 分别定义恒等态射与复合, 验证复合仍然满足所要求的交换条件(commutativity condition), 并验证范畴公理.

    \item[(b)] 在 $\mathbf{Set}/A$ 中, 取
      \[
        A=\{0,1\},\qquad X=\{1,2,3\},\qquad Y=\{u,v,w\},
      \]
      并定义
      \[
        p(1)=p(2)=0,\quad p(3)=1,\qquad q(u)=0,\quad q(v)=q(w)=1.
      \]
      列出 $\Hom_{\mathbf{Set}/A}(p,q)$ 中的全部态射. 对象 $p$ 与 $q$ 是否同构? 比较这个问题与``集合 $X$ 和 $Y$ 是否同构''.

    \item[(c)] (选做)尝试想象有如下结果, 如何理解和严谨的定义和说明:
      \[
        (\mathcal C/A)\op\cong A/\mathcal C\op,\qquad
        (A/\mathcal C)\op\cong\mathcal C\op/A.
      \]
  \end{enumerate}
\end{problem}

\begin{solution}
	(a) 事实上, 由 (c) 中的结果, 只需要考虑切片范畴 $\mathcal{C}/A$.

	对于 $p\in \Hom_{\mathcal{C}}(X,A)$, 定义其恒等态射
	\[ \id_p^{\mathcal{C}/A} = \id_X^{\mathcal{C}}. \]
	对于 $\mathcal{C}/A$ 中的 $p \xrightarrow{u} q \xrightarrow{v} r$, 定义 $u,v$ 在 $\mathcal{C}/A$ 中的复合即为 $\mathcal{C}$ 中的复合.下面来验证范畴公理.

	先验证恒等态射是被正确定义的. 给定 $\mathcal{C}/A$ 的对象 $p\in \Hom_{\mathcal{C}}(X,A)$ 与 $q\in \Hom_{\mathcal{C}}(Y,A)$. 对于任意的 $u \in \Hom_{\mathcal{C}/A}(p,q)$ 以及 $v \in \Hom_{\mathcal{C}/A}(q,p)$, 根据复合的定义, 有
	\[ u \circ \id_p^{\mathcal{C}/A} = u \circ \id_X^{\mathcal{C}} = u,\quad
	\id_p^{\mathcal{C}/A} \circ v = \id_Y^{\mathcal{C}} \circ v = v. \]
	因此 $\mathcal{C}/A$ 中的恒等态射是被正确定义的. 

	再验证复合满足交换条件. 对于 $\mathcal{C}/A$ 的对象 $p\in \Hom_\mathcal{C} (X,A)$, $q\in \Hom_\mathcal{C} (Y,A)$ 与 $r\in \Hom_{\mathcal{C}}(Z,A)$, 取出 $\mathcal{C}/A$ 的三个态射 $p \xrightarrow{u} q \xrightarrow{v} r$. 则 $q \circ u = p$, $r \circ v = q$, 因此
	\[ r \circ (v \circ u) = (r \circ v) \circ u = q \circ u =p. \]
	从而 $v\circ u \in \Hom_{\mathcal{C}/A}(p,r)$. 

	接下来验证结合律. 对于 $\mathcal{C}/A$ 中的三个态射 $p \xrightarrow{u} q \xrightarrow{v} r \xrightarrow{w} s$, 由于它们在 $\mathcal{C}/A$ 中的复合即为在 $\mathcal{C}$ 中的复合, 因此结合律自然成立.

	(b) $\Hom_{\mathbf{Set}/A}(p,q)$ 中有两个态射: 
	\[ 
		\begin{array}{c|ccc}
			& 1 & 2 & 3 \\ \hline
			f_1(\cdot) & u & u & v \\
			f_2(\cdot) & u & u & w
		\end{array}
	\]
	不难验证 $q \circ f_{1,2} = p$. 

	对象 $p,q$ 不是同构的, 因为上面给出的 $f_1$ 与 $f_2$ 都没有逆态射. 而集合 $X$ 与 $Y$ 是同构的, 因为 $|X| = |Y| = 3$. 

	(c) 先来说明第一个同构关系, 即 $(\mathcal{C}/A)\op \cong A/\mathcal{C}\op$. 对于 $\mathcal{C}/A$ 中的一个对象 $p \in \Hom_{\mathcal{C}}(X,A)$, 将它与 $A/\mathcal{C}\op$ 中的对象 $p\op \in \Hom_{\mathcal{C}\op}(A,X)$ 对应; 对于 $\mathcal{C}/A$ 中的态射 $f \in \Hom_{\mathcal{C}/A} (p,q)$, 将它与 $A/\mathcal{C}\op$ 中的态射 $f\op \in \Hom_{A/\mathcal{C}\op}(q\op,p\op)$ 对应. 这样的定义满足所要求的交换关系, 因为若 $q \circ f = p$, 则 $f\op \circ q\op = p\op$, 即 $f\op \in \Hom_{A/\mathcal{C}\op}(q\op,p\op)$.

	对于第二个同构关系, 注意到可以直接使用第一个同构关系, 结合 $(\mathcal{C}\op)\op \cong \mathcal{C}$ 来证明: 
	\[ (A/\mathcal{C})\op 
	\cong (A/(\mathcal{C}\op)\op)\op
	\cong ((\mathcal{C}\op/A)\op)\op
	\cong \mathcal{C}\op/A. \]
\end{solution}



\begin{problem}
  {\heiti 反变函子}(\textbf{contravariant functor}){\heiti 的定义与集合上的例子}. 定义. 从范畴 $\mathcal C$ 到范畴 $\mathcal D$ 的一个反变函子 $F$ 由以下数据组成: 对每个对象 $X\in\operatorname{Ob}(\mathcal C)$, 指定对象 $F(X)\in\operatorname{Ob}(\mathcal D)$; 对每对对象 $X,Y$, 指定集合映射
  \[
    F_{X,Y}:\Hom_{\mathcal C}(X,Y)\longrightarrow
    \Hom_{\mathcal D}(F(Y),F(X)).
  \]
  将 $F_{X,Y}(f)$ 简记为 $F(f)$. 这些数据必须满足
  \[
    F(\id_X)=\id_{F(X)}
  \]
  以及: 对每一对可复合态射 $X\xrightarrow{f}Y\xrightarrow{g}Z$, 有
  \[
    F(g\circ f)=F(f)\circ F(g)\qquad\text{于 }\Hom_{\mathcal D}(F(Z),F(X)).
  \]

  \begin{enumerate}
    \item[(a)] 证明: 给出上述意义下的反变函子, 等价于给出一个协变函子
      \[
        \widetilde F:\mathcal C\op\longrightarrow\mathcal D,
      \]
      其中 $\widetilde F(X)=F(X)$, $\widetilde F(f\op)=F(f)$. 逐项比较两种表述中的恒等态射公理和复合公理.

    \item[(b)] 对集合 $X$, 记其幂集(power set)为 $\mathcal P(X)$. 对映射 $f:X\to Y$, 用逆像(inverse image)定义
      \[
        \mathcal P^*(f\op):\mathcal P(Y)\longrightarrow\mathcal P(X),\qquad
        B\longmapsto f^{-1}(B).
      \]
      证明这定义一个函子 $\mathcal P^*:\mathbf{Set}\op\to\mathbf{Set}$, 即从 $\mathbf{Set}$ 到 $\mathbf{Set}$ 的一个反变函子.

      具体取
      \[
        X=\{1,2,3\},\qquad Y=\{a,b\},\qquad f(1)=f(2)=a,\quad f(3)=b.
      \]
      计算 $\mathcal P^*(f\op)$ 在 $\mathcal P(Y)$ 的每个元素上的值.

    \item[(c)] 同样的对象赋值 $X\mapsto\mathcal P(X)$, 配上直接像(direct image)映射
      \[
        \mathcal P_*(f):\mathcal P(X)\longrightarrow\mathcal P(Y),\qquad B\longmapsto f(B),
      \]
      是否定义一个协变函子 $\mathcal P_*:\mathbf{Set}\to\mathbf{Set}$? 验证你的结论.
  \end{enumerate}
\end{problem}

\begin{solution}
	(a) 先比较两种表述中的恒等态射公理. 对于 $\mathcal{C}$ 的对象 $X$, 有
	\[ F(\id_X) = \id_{F(X)} \Longleftrightarrow \tilde{F}(\id_X\op) = F(\id_X) = \id_{F(X)} = \id_{\tilde{F}(X)}. \]
	再比较复合公理. 对于 $\mathcal{C}$ 中的两个态射 $X \xrightarrow{f} Y \xrightarrow{g} Z$, 我们有
	\begin{gather*}
		F(g\circ f) = F(f) \circ F(g) \\
		\Longleftrightarrow \\
		\tilde{F}(f\op \circ g\op) 
		= \tilde{F}((g\circ f)\op) 
		= F(g\circ f) = F(f) \circ F(g)
		=\tilde{F}(f\op) \circ \tilde{F}(g\op).
	\end{gather*}

	(b) 我们来验证 $\mathcal{P}^*$ 满足反变函子的两条公理. 对于 $\id_X^{\mathbf{Set}}$, 我们有
	\[ \mathcal{P}^*(\id_X^{\mathbf{Set}\op}): \mathcal{P}(X) \to \mathcal{P}(X), \quad B \mapsto (\id_X)^{-1}(B) = B = \id_{\mathcal{P}(X)} (B). \]
	因此 $\mathcal{P}^*(\id_X^{\mathbf{Set}\op}) = \id_{\mathcal{P}(X)}$. 

	对于 $\mathbf{Set}$ 中的态射 $X \xrightarrow{f} Y \xrightarrow{g} Z$, 根据定义, 
	\[ \mathcal{P}^*(f\op \circ g\op) = \mathcal{P}^*((g\circ f)\op) = (g \circ f)^{-1} = f^{-1} \circ g^{-1} = \mathcal{P}^*(f\op) \circ \mathcal{P}^*(g\op). \]

	因此 $\mathcal{P}^*$ 给出了 $\mathbf{Set}\op \to \mathbf{Set}$ 的一个协变函子. 

	对于题目中给出的例子, 计算如下: 
	\[ \begin{array}{c|cccc}
		 B & \varnothing & \{a\} & \{b\} & \{a,b\} \\ \hline
		 \mathcal{P}^*(f\op)(B) & \varnothing & \{1,2\} & \{3\} &\{1,2,3\}
	\end{array} \]

	(c) 我们来验证 $\mathcal{P}_*$ 满足协变函子的公理. 对于一个集合 $X$, 我们有
	\[ \mathcal{P}_*(\id_X): \mathcal{P}(X) \to \mathcal{P}(X), \quad B \mapsto \id_X(B) = B = \id_{\mathcal{P}_*(X)}(B), \]
	因此 $\mathcal{P}_*(\id_X) = \id_{\mathcal{P}_*(X)}$. 

	对于集合之间的映射 $X \xrightarrow{f} Y \xrightarrow{g} Z$, 我们有
	\[ \mathcal{P}_*(g\circ f) = g \circ f = \mathcal{P}_*(g) \circ \mathcal{P}_*(f). \]

	因此 $\mathcal{P}_*: \mathbf{Set} \to \mathbf{Set}$ 是一个协变函子.
\end{solution}

        \begin{array}{c|c|c}
          F & G & \text{合法的复合}\\ \hline
          F:\mathcal C\op\to\mathcal D & G:\mathcal D\to\mathcal E & G\circ F:\mathcal C\op\to\mathcal E\\
          F:\mathcal C\to\mathcal D & G:\mathcal D\op\to\mathcal E & G\circ F\op:\mathcal C\op\to\mathcal E\\

        \end{array}
      \]

\begin{solution}
	(a) 我们来验证 $G\circ F$ 满足函子的两条公理: 对于 $\mathcal{C}$ 中的对象 $X$, 
	\[ (G\circ F)(\id_X) = G(F(\id_X)) = G(\id_{F(X)}) = \id_{G(F(X))} = \id_{(G\circ F)(X)}. \]
	对于 $\mathcal{C}$ 中的态射 $X \xrightarrow{f} Y \xrightarrow{g} Z$,
	\[ (G\circ F)(g\circ f) = G(F(g\circ f)) = G(F(g) \circ F(f))
	= G(F(g)) \circ G(F(f)) = (G\circ F)(g) \circ (G\circ F)(f). \]

	(b) 对于 $\mathcal{C}$ 的一个对象 $X$, 有
	\[ (F\circ \id_\mathcal{C}) (X) = F(\id_\mathcal{C}(X)) = F(X),\quad 
	(\id_\mathcal{D} \circ F)(X) = \id_\mathcal{D}(F(X)) = F(X); \]
	并且
	\[ (H\circ(G\circ F))(X) = H((G\circ F)(X)) = H(G(F(X)))
	=(H\circ G)(F(X)) = ((H\circ G)\circ F)(X). \]
	对于 $\mathcal{C}$ 中的态射 $X \xrightarrow{f}Y$, 有
	\[ (F\circ \id_\mathcal{C}) (f) = F(\id_\mathcal{C}(f)) = F(f),\quad 
	(\id_\mathcal{D} \circ F)(f) = \id_\mathcal{D}(F(f)) = F(f); \]
	并且
	\[ (H\circ(G\circ F))(f) = H((G\circ F)(f)) = H(G(F(f)))
	=(H\circ G)(F(f)) = ((H\circ G)\circ F)(f). \]

	(c) 对于 $\mathcal{A}$ 的一个对象 $X$, 有 
	\[ K\op(\id_X\op{}^{\mathcal{A}\op}) = K(\id_X^{\mathcal{A}})\op = (\id_{K(X)}^{\mathcal{B}})\op, \]
	对于 $\mathcal{A}$ 中的两个态射 $X \xrightarrow{f} Y \xrightarrow{g} Z$, 有
	\[ K\op(f\op \circ g\op) = K\op((g\circ f)\op) = K(g\circ f)\op
	=(K(g) \circ K(f))\op = K(f)\op \circ K(g)\op
	=K\op(f\op)\circ K\op(g\op). \]
	因此 $K\op$ 是函子.

	接下来, 我们来验证表格中的每一个函子的复合都是合法的. 

	第一个复合为 $\mathcal{C}\op \xrightarrow{F} \mathcal{D} \xrightarrow{G} \mathcal{E}$, 其中 $F$ 的陪域等于 $G$ 的定义域, 合法; 
	第二个复合为 $\mathcal{C}\op \xrightarrow{F\op} \mathcal{D}\op \xrightarrow{G} \mathcal{E}$, 由相反范畴的构造, $F\op$ 的陪域与 $G$ 的定义域均为 $\mathcal{D}\op$, 故为合法复合;
	第三个复合为 $\mathcal{C} \xrightarrow{F\op} \mathcal{D}\op \xrightarrow{G} \mathcal{E}$, 由相反范畴的构造, $F\op$ 的陪域与 $G$ 的定义域均为 $\mathcal{D}\op$, 故为合法复合.

	从复合函子的定义域来看, 前两个复合函子相对于 $\mathcal{C}$ 为逆变函子, 而第三个符合函子相对于 $\mathcal{C}$ 为协变函子.
\end{solution}



\begin{problem}
  {\heiti 单态射}(\textbf{monomorphism}){\heiti 与满态射}(\textbf{epimorphism}){\heiti : 定义和基本性质}. 定义. 设 $f:X\to Y$ 是范畴 $\mathcal C$ 中的一个态射.

  \begin{itemize}
    \item 若对任意对象 $T$ 和任意两个态射 $a,b:T\to X$, 都有
      \[
        f\circ a=f\circ b\qquad\Longrightarrow\qquad a=b,
      \]
      则称 $f$ 为单态射(monomorphism, 简称 mono).

    \item 若对任意对象 $T$ 和任意两个态射 $a,b:Y\to T$, 都有
      \[
        a\circ f=b\circ f\qquad\Longrightarrow\qquad a=b,
      \]
      则称 $f$ 为满态射(epimorphism, 简称 epi).
  \end{itemize}

  证明以下结论:
  \begin{enumerate}
    \item[(a)] 若 $f$ 有左逆, 证明 $f$ 是单态射; 若 $f$ 有右逆, 证明 $f$ 是满态射. 由此证明每个同构都是单态射和满态射.

    \item[(b)] 对可复合的态射 $X\xrightarrow{f}Y\xrightarrow{g}Z$, 证明:
      \begin{enumerate}
        \item[(i)] 若 $f,g$ 都是单态射, 则 $g\circ f$ 是单态射; 若 $f,g$ 都是满态射, 则 $g\circ f$ 是满态射.
        \item[(ii)] 若 $g\circ f$ 是单态射, 则 $f$ 是单态射; 若 $g\circ f$ 是满态射, 则 $g$ 是满态射.
      \end{enumerate}

    \item[(c)] 用上述定义证明单态射与满态射的对偶关系:
			\begin{gather*}
        f\text{ 在 }\mathcal C\text{ 中是单态射}\qquad\Longleftrightarrow\qquad
        f\op\text{ 在 }\mathcal C\op\text{ 中是满态射},\\
        f\text{ 在 }\mathcal C\text{ 中是满态射}\qquad\Longleftrightarrow\qquad
        f\op\text{ 在 }\mathcal C\op\text{ 中是单态射}.
			\end{gather*}
  \end{enumerate}
\end{problem}

\begin{solution}
	(a) 若 $f$ 存在左逆, 即若存在 $g: Y \to X$, 使得 $g\circ f = \id_X$, 则对任意的对象 $T$ 和两个态射 $a,b: T\to X$, 若有 $f \circ a = f \circ b$, 则有
	\[ a = \id_X \circ a = g \circ f \circ a = g \circ f \circ b = \id_X \circ b = b. \]
	因此 $f$ 是单态射. 若 $f$ 存在右逆, 则由类似的讨论可以证明 $f$ 为满态射. 而同构同时存在左逆和右逆, 因此每一个同构都同时为单态射和满态射.

	(b.i) 若 $f,g$ 均为单态射, 则对于任意的态射 $a,b:T\to X$, 有
	\[ (g\circ f)\circ a = (g\circ f)\circ b \implies f \circ a = f\circ b \implies a=b. \]
	因此 $g\circ f$ 为单态射. 同理可以证明, 若 $f,g$ 均为满态射, 则 $g\circ f$ 也为满态射.

	(b.ii) 若 $g\circ f$ 是单态射, 则对于任意的态射 $a,b:T\to X$, 有
	\[ f \circ a = f \circ b \implies g \circ f \circ a = g \circ f \circ b \implies a = b. \]
	因此 $f$ 是单态射. 同理可以证明, 若 $g\circ f$ 为满态射, 则 $g$ 为满态射.

	(c) 事实上, 
	\[ \begin{array}{cc}
		& f \text{在} \mathcal{C} \text{中是单态射} \\
		\Longleftrightarrow & \forall\ a,b:T \to X, \quad (f \circ a = f \circ b \implies a=b) \\
		\Longleftrightarrow & \forall\ a\op, b\op: X\to T, \quad (a\op \circ f\op = b\op \circ f\op \implies a\op = b\op) \\
		\Longleftrightarrow & f\op \text{在} \mathcal{C}\op \text{中是满态射}.
	\end{array} \]
	同理可以证明另一个命题.
\end{solution}



\begin{problem}
  \begin{enumerate}
    \item[(a)] 在 $\mathbf{Set}$ 中证明
			\begin{gather*}
        f\text{ 是单态射}\qquad\Longleftrightarrow\qquad f\text{ 是单射(injection)},\\
        f\text{ 是满态射}\qquad\Longleftrightarrow\qquad f\text{ 是满射(surjection)}.
			\end{gather*}

      提示: 分别考虑以单点集为定义域、以两点集为陪域的测试映射.

      由此证明: 在 $\mathbf{Set}$ 中, 一个态射同时是单态射和满态射, 当且仅当它是双射(bijection), 也当且仅当它是同构.

    \item[(b)] 令 $\mathcal I$ 是偏序 $0<1$ 对应的范畴, 即除恒等态射外, 只有一个态射 $u:0\to1$. 证明 $u$ 同时是单态射和满态射, 但不是同构.

      进一步证明: 在任意偏序对应的范畴中, 每个态射都是单态射和满态射. 解释为什么这个反例与上一小题的结论不矛盾.
  \end{enumerate}
\end{problem}

\begin{solution}
	(a) 先来证明第一个命题, 即 $f$ 是单态射 $\Longleftrightarrow$ $f$ 是单射. 假设 $f:X\to Y$.

	``$\implies$'': 取 $T = \{e\}$. 若 $f$ 不是单射, 则 $X$ 中存在不相等的元素 $a\neq b$, 满足 $f(a)=f(b)$. 考虑两个映射 $g_{1,2}: T\to X$, $g_1(e) = a$, $g_2(e) = b$. 则 $(f \circ g_1)(e) = f(a) = f(b) = (f \circ g_2)(e)$, 即 $f \circ g_1 = f\circ g_2$, 但 $g_1 \neq g_2$, 这与 $f$ 为单态射矛盾. 因此 $f$ 为单射.

	``$\impliedby$'': 若 $f$ 为单射, 则 $f$ 存在左逆映射, 从而 $f$ 是单态射.

	再证明第二个命题, 即 $f$ 是满态射 $\Longleftarrow$ $f$ 是满射.

	``$\implies$'': 取 $T = \{x,y\}$. 若 $f$ 不为满射, 则集合 $Y-f(X)$ 非空. 考虑两个映射 $g_{1,2}: Y \to T$, 满足
	\[ g_1(e) = x,\ \forall\ e\in Y;\quad 
	g_2(e) = \begin{cases}
		x, & e \in f(X),\\
		y, & e \in Y-f(X).
	\end{cases} \]
	由 $Y-f(X) \neq \varnothing$, 知 $g_1 \neq g_2$; 而对于任意的 $e\in X$, 由于 $f(e)\in f(X)$, 因此
	\[ g_2\circ f(e) = x = g_1 \circ f(e), \]
	即 $g_1 \circ f = g_2 \circ f$, 这与 $f$ 为满态射矛盾. 因此 $f$ 为满射.

	``$\impliedby$'': 若 $f$ 为满射, 则 $f$ 存在右逆映射, 从而 $f$ 是满态射.

	因此, 在 $\mathbf{Set}$ 中, 一个态射同时是单态射和满态射, 当且仅当这个态射作为集合的映射同时是单射和满射, 当且仅当它是双射. 而集合之间的双射即为同构, 因此上面的命题也等价于它是同构.

	(b) 先证明 $u$ 是单态射. 若存在态射 $a,b$ 使得 $u \circ a = u \circ b$, 则为了使复合有意义, 必须有 $a=b= \id_0$, 因此 $a=b$, 从而 $u$ 是单态射. 同理可以证明 $u$ 是满态射. 但是 $u$ 不是同构, 因为它既没有左逆也没有右逆.

	一般地, 设 $P$ 为一个偏序集, $\mathcal{C}_P$ 为其对应的范畴. 对于每一个态射 $f: X\to Y$, 若存在两个态射 $g_1,g_2: T \to X$, 使得 $f \circ g_1 = f\circ g_2$, 则由于 $\Hom_{\mathcal{C}_P}(T,X)$ 中至多只有一个元素, 因此必定有 $g_1 = g_2$, 从而 $f$ 是单态射. 同理, $f$ 为满态射. 

	这不与 (a) 中的结论矛盾, 因为显然对于任意的偏序集 $P$, $\mathcal{C}_P$ 均不与 $\mathbf{Set}$ 同构, 因为 $\mathcal{C}_P$ 中任意两个对象之间至多只有一个态射, 而 $\mathbf{Set}$ 中的两个对象之间可以有多于一个态射.
\end{solution}



\begin{problem}
  \textbf{Hom} {\heiti 函子}(\textbf{Hom functors}). 固定范畴 $\mathcal C$ 中的对象 $A$.

  \begin{enumerate}
    \item[(a)] 严格的给出定义 $\Hom_{\mathcal C}(A,-):\mathcal C\to\mathbf{Set}$, 满足在态射 $f:X\to Y$ 上的作用为
      \[
        \Hom_{\mathcal C}(A,X)\longrightarrow\Hom_{\mathcal C}(A,Y),\qquad
        u\longmapsto f\circ u.
      \]
      严格的给出定义 $\Hom_{\mathcal C}(-,A):\mathcal C\op\to\mathbf{Set}$ 满足在 $f\op:Y\to X$ 上的作用为
      \[
        \Hom_{\mathcal C}(Y,A)\longrightarrow\Hom_{\mathcal C}(X,A),\qquad
        v\longmapsto v\circ f.
      \]
      验证前者是协变函子, 后者给出从 $\mathcal C$ 到 $\mathbf{Set}$ 的反变函子. 这两种作用分别称为后复合 (postcomposition) 和预复合 (precomposition) .

    \item[(b)] 给定 $f:X\to Y$, 对每个对象 $T$, 记
      \[
        \begin{aligned}
          f_{*,T}&:\Hom(T,X)\longrightarrow\Hom(T,Y),
          & a&\longmapsto f\circ a,\\
          f_T^*&:\Hom(Y,T)\longrightarrow\Hom(X,T),
          & b&\longmapsto b\circ f.
        \end{aligned}
      \]
      证明
      \[
        \begin{aligned}
          f\text{ 是单态射}&\qquad\Longleftrightarrow\qquad f_{*,T}\text{ 对每个 }T\text{ 都是单射},\\
          f\text{ 是满态射}&\qquad\Longleftrightarrow\qquad f_T^*\text{ 对每个 }T\text{ 都是单射}.
        \end{aligned}
      \]
  \end{enumerate}
\end{problem}

\begin{solution}
	(a) 简记 $\Hom_{\mathcal{C}}(A,\cdot) = F$, $\Hom_{\mathcal{C}}(\cdot,A) = G$. 我们给出 $F, G$ 的具体定义: 

	$F: \mathcal{C} \to \mathbf{Set}$, 满足对于 $\mathcal{C}$ 的任意对象 $X$, 以及任意态射 $f: X\to Y$, 定义 $F(X) := \Hom_{\mathcal{C}}(A,X)$, 并且
	\[ F(f): \Hom_{\mathcal C}(A,X) \to \Hom_{\mathcal C}(A,Y),\quad u\mapsto f\circ u. \]

	$G: \mathcal{C}\op \to \mathbf{Set}$, 满足对于 $\mathcal{C}\op$ 的任意对象 $X$ 以及任意态射 $f\op: Y\to X$, 定义 $G(X) := \Hom_{\mathcal{C}}(X,A)$, 并且
	\[ G(f\op): \Hom_{\mathcal{C}}(Y,A) \to \Hom_{\mathcal{C}}(X,A), \quad u \mapsto u \circ f. \]

	现在来验证 $F$ 是协变函子. 首先, 对于 $\mathcal{C}$ 中的恒等态射 $\id_X$, 根据 $F$ 的定义, 
	\[ F(\id_X^{\mathcal{C}}): \Hom_{\mathcal{C}}(A,X) \to \Hom_{\mathcal{C}}(A,X), \quad u \mapsto \id_X \circ u = u = \id_{F(X)}^{\mathbf{Set}}(u). \]
	因此 $F(\id_X^{\mathcal{C}}) = \id_{F(X)}^{\mathbf{Set}}$.

	其次, 对于 $\mathcal{C}$ 中的态射 $X \xrightarrow{f} Y \xrightarrow{g} Z$, 有
	\begin{align*}
		F(g\circ f): 
		&\Hom_{\mathcal{C}}(A,X) \to \Hom_{\mathcal{C}}(A,Z), \\
		&u \mapsto g \circ f \circ u = g \circ F(f)(u) = F(g)[F(f)(u)] = (F(g) \circ F(f))(u). 
	\end{align*}

	因此 $F(g\circ f) = F(g) \circ F(f)$.

	再来验证 $G$ 是 $\mathcal{C}$ 到 $\mathbf{Set}$ 的逆变函子. 首先, 对于 $\mathcal{C}\op$ 中的恒等态射 $\id_X\op = \id_X$, 有
	\[ G(\id_X\op{}^{\mathcal{C}\op}): \Hom_{\mathcal{C}}(X,A) \to \Hom_{\mathcal{C}}(X,A), \quad u \mapsto u \circ \id_X\op = u = \id_{G(X)}^{\mathbf{Set}}(u). \]
	因此 $G(\id_X\op) = \id_{G(X)}^{\mathbf{Set}}$.

	其次, 对于 $\mathcal{C}\op$ 中的态射 $Z \xrightarrow{g\op} Y \xrightarrow{f\op} X$, 有
	\begin{align*}
		G(f\op \circ g\op): 
		&\Hom_{\mathcal{C}}(Z,A) \to \Hom_{\mathcal{C}}(X,A), \\
		&u \mapsto u \circ (f\op \circ g\op)\op = u \circ g \circ f \\
		&\qquad =(G(g\op)(u)) \circ f = G(f\op)[G(g\op)(u)] =(G(f\op) \circ G(g\op))(u).
	\end{align*}
	因此 $G(f\op \circ g\op) = G(f\op) \circ G(g\op)$.

	(b) 先证明第一个命题, 即 $f$ 是单态射 $\Leftrightarrow$ $f_{*,T}$ 对每一个 $T$ 都是单射. 事实上,
	\[ \begin{array}{cc}
		 & f \text{ 为单态射} \\
		\Leftrightarrow & \forall\ a,b: T \to X, \quad (f \circ a = f \circ b \implies a = b) \\
		\Leftrightarrow & \forall\ a,b: T \to X, \quad (f_{*,T}(a) = f_{*,T}(b) \implies a = b) \\
		\Leftrightarrow & f_{*,T} \text{ 为单射}.
	\end{array} \]
	第二个命题的证明与之同理.
\end{solution}



\begin{problem}
  设 $F:\mathcal C\to\mathcal D$ 是函子. 如果对每对对象 $X,Y$, 映射
  \[
    \Hom_{\mathcal C}(X,Y)\longrightarrow
    \Hom_{\mathcal D}(F(X),F(Y))
  \]
  都是单射, 则称 $F$ 为{\heiti 忠实函子}(\textbf{faithful functor}).

  \begin{enumerate}
    \item[(a)] 证明任意函子都保持同构, 并且对同构 $f$ 有
      \[
        F(f^{-1})=F(f)^{-1}.
      \]

    \item[(b)] 证明忠实函子反映单态射和满态射, 即
			\begin{gather*}
        F(f)\text{ 是单态射}\qquad\Longrightarrow\qquad f\text{ 是单态射},\\
        F(f)\text{ 是满态射}\qquad\Longrightarrow\qquad f\text{ 是满态射}.
			\end{gather*}
  \end{enumerate}
\end{problem}

\begin{solution}
	(a) 设 $\mathcal{C}$ 中的态射 $f: X\to Y$ 是同构, 即存在 $f^{-1}: Y\to X$, 满足 $f \circ f^{-1} = \id_Y$, $f^{-1} \circ f = \id_X$.

	注意到
	\begin{gather*}
		F(f) \circ F(f^{-1}) = F(f \circ f^{-1}) = F(\id_Y^{\mathcal{C}}) = \id_{F(Y)}^{\mathcal{D}}, \\
		F(f^{-1}) \circ F(f) = F(f^{-1} \circ f) = F(\id_X^{\mathcal{C}}) = \id_{F(X)}^{\mathcal{D}}.
	\end{gather*}
	上面的结果表明 $F(f^{-1})$ 是 $F(f)$ 的逆态射, 因此 $F(f)$ 仍然是同构, 并且 $F(f)^{-1} = F(f^{-1})$.

	(b) 由于 $F$ 此时为忠实函子, 因此对于 $\mathcal{D}$ 中的态射 $f$, 只要它在 $\mathcal{C}$ 中有``原像'', 那么 $F^{-1}(f)$ 是有意义的. 取定 $\mathcal{C}$ 中的态射 $f:X\to Y$. 注意到 (我们默认式子中出现的表达式都是有意义的) 
	\[ \begin{array}{cc}
		 & F(f) \text{ 是单态射} \\
		\implies & \forall\ a,b: T\to F(X), \quad 
		(F(f) \circ a = F(f) \circ b \implies a = b) \\
		\implies & \forall\ a,b: T\to F(X), \quad 
		(f \circ F^{-1}(a) = f \circ F^{-1}(b) \implies F^{-1}(a) = F^{-1}(b)) \\
		\implies & \forall\ a',b': F^{-1}(T)\to X, \quad 
		(f \circ a' = f \circ b' \implies a' = b') \\
		\implies & f \text{ 为单态射}.
	\end{array} \]

	第二个命题的证明与之同理.
\end{solution}
