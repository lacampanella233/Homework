\begin{problem}
(30pts) Consider a spherically symmetric planet of radius $R$ and uniform mass density $\rho$.
\begin{enumerate}[label=(\alph*)]
	\item (20pts) Find the potential energy of a particle of mass $m$ at a position $\bo{r}$ in the interior of the planet.
	\item (10pts) Compute the force associated with this potential. The force should agree with the expression computed in class using Gauss' law.
\end{enumerate}
\end{problem}

\begin{solution}
  (a) By the spherical symmetry of the problem, the potential energy of the particle at a position $\bo{r}$ in the interior of the planet only depends on the distance $r$ from the center of the planet. Then the potential energy can be written as \( \varphi(\bo{r})=\varphi(r) \).\par
  \( \varphi \) can be solvedd by the Poisson's equation in the spherical coordinates:
  \[ \nabla^2\varphi(\bo{r})= 4\pi G \rho\ (r<R) 
  \implies \frac{1}{r^2}\frac{\d}{\d r}\left( r^2\frac{\d \varphi(r)}{\d r}\right)=4\pi G\rho 
  \implies \varphi(r) = \frac{2}{3}\pi G \rho r^2+C. \]
  (b) The force can be computed by
  \[ \bo{F}=-\nabla\varphi(\bo{r})= -\frac{\d \varphi(r)}{\d r}\bo{e}_r = -\frac{4}{3}\pi G \rho r \bo{e}_r. \]
\end{solution}



\begin{problem}
(20pts) Consider a uniform, i.e.\ constant, electric field $\bo{E}$.
\begin{enumerate}[label=(\alph*)]
	\item (10pts) Find the potential energy of a particle with charge $q$ in this electric field.
	\item (10pts) Using the expression above, find the potential energy of a dipole with moment $\bo{p}$ in this electric field.
\end{enumerate}
\end{problem}

\begin{solution}
  (a) Select the coordinate system such that the electric field is along the $x$-axis, i.e.\ $\bo{E}=E\bo{e}_x$. Then the potential energy of a particle with charge $q$ at \( \bo{r}=(x,y,z) \) is given by
  \[ \mathcal{E} = -q\bo{E} \cdot \bo{r} = -qEx. \]\par
  (b) An electric dipole with moment \( \bo{p} \) can be considered as two opposite charges \( +q \) and \( -q \) separated by a small distance \( \bo{d} \), such that \( \bo{p}=q\bo{d} \). The potential energy of the dipole in the electric field can be computed by
  \[ \mathcal{E} = -q (\bo{r}+\bo{d})\cdot\bo{E}-(-q)\bo{r}\cdot\bo{E}= -\bo{p} \cdot \bo{E}. \]
\end{solution}

\begin{enumerate}[label=(\alph*)]
	\item (10pts) Sketch the electric field lines.
	\item (10pts) Write the electric field $\bo{E}$ as a function of the radial coordinate $\bo{r}$ (make sure to indicate the origin of your coordinate system).
	\item (15pts) The electric field $\bo{E}$ at a distance $r \gg |d|$ can be expanded in powers of $1/r$. The leading term in the expansion falls off as $1/r^n$ for some integer $n$. Find the leading term. Explain why we should expect this result.
	\item (15pts) Compute the next term in the expansion of the electric field at a distance $r \gg |d|$, that is, the term that goes like $1/r^{n+1}$, where $n$ was determined above. Explain why we should expect this result.
\end{enumerate}
\end{problem}

\begin{solution}
  (a) The diagram below shows the electric feild lines and the isolines of the electric potential.
  \begin{figure}[H]
    \centering
    \includegraphics[width=0.4\textwidth]{PLOT.pdf}
  \end{figure}\par
  (b) We use the coordinate system with the origin at the position of the negative charge, and the positive charges are located at \( \bo{d} \) and \( -\bo{d} \). The electric field at a point \( \bo{r} \) can be computed by superposition:
  \[ \bo{E}(\bo{r})=q\frac{\bo{r}-\bo{d}}{|\bo{r}-\bo{d}|^3} + q\frac{\bo{r}+\bo{d}}{|\bo{r}+\bo{d}|^3}- q\frac{\bo{r}}{|\bo{r}|^3}. \]\par
  (c) Denote \( \hat{\bo{r}}=\bo{r}/\abs{\bo{r}} \) and \( r=\abs{\bo{r}} \). Then
  \begin{align*}
    \bo{E}(\bo{r})
    &=\frac{1}{r^2}\left( q\frac{\hat{\bo{r}}-\bo{d}/r}{\abs{\hat{\bo{r}}-\bo{d}/r}^3}+ q\frac{\hat{\bo{r}}+\bo{d}/r}{\abs{\hat{\bo{r}}+\bo{d}/r}^3}- q\frac{\hat{\bo{r}}}{\abs{\hat{\bo{r}}}^3} \right)\\
    &\approx\frac{q}{r^2}\left\{ \left( \hat{\bo{r}}-\frac{1}{r}\bo{d} \right)\left( 1-3\hat{\bo{r}}\cdot\frac{\bo{d}}{r} \right)+\left( \hat{\bo{r}}+\frac{1}{r}\bo{d} \right)\left( 1+3\hat{\bo{r}}\cdot\frac{\bo{d}}{r} \right)-\hat{\bo{r}} \right\}\\
    &\approx\frac{q}{r^2}\left( \hat{\bo{r}}+\frac{6\hat{\bo{r}}\cdot\bo{d}}{r^2}\bo{d} \right).
  \end{align*}
\end{solution}
